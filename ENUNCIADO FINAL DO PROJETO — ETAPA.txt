🛡️ 2. Parte 1 — Blindagem Final da API (Middlewares Essenciais)
Para garantir que a aplicação não quebre durante a apresentação ao vivo, implemente os 3 middlewares essenciais dentro da pasta src/middlewares/:
A. Validação e Sanitização de Dados (validarDados.js)
Crie um middleware na pasta src/middlewares/ para validar os campos obrigatórios antes de salvar no banco via Prisma (adaptando os nomes dos campos para o tema do seu grupo):// Exemplo para Post (adapte para o tema do seu grupo)
function validarPost(req, res, next) {
const { titulo, conteudo, autorId } = req.body;

if (!titulo || typeof titulo !== 'string' || titulo.trim().length < 5) {
return res.status(400).json({
erro: "O título é obrigatório e deve ter no mínimo 5 caracteres."
});
}

if (!conteudo || typeof conteudo !== 'string' || conteudo.trim() === "") {
return res.status(400).json({
erro: "O conteúdo não pode ficar em branco."
});
}

if (req.method === 'POST') {
const idNumerico = Number(autorId);
if (!autorId || isNaN(idNumerico) || idNumerico <= 0) {
return res.status(400).json({
erro: "O 'autorId' é obrigatório e deve ser um número válido."
});
}
req.body.autorId = idNumerico;
}

// Sanitização: remove espaços extras nas pontas
req.body.titulo = titulo.trim();
req.body.conteudo = conteudo.trim();

next();
}

module.exports = validarPost;

B. Validação de ID na URL (validarId.js)
Protege as rotas GET /:id, PUT /:id e DELETE /:id contra parâmetros que não sejam números inteiros positivos:// src/middlewares/validarId.js
function validarId(req, res, next) {
const id = Number(req.params.id);

if (isNaN(id) || !Number.isInteger(id) || id <= 0) {
return res.status(400).json({
erro: "ID inválido na URL. Utilize um número inteiro positivo."
});
}

next();
}

module.exports = validarId;

C. Middleware Global de Auditoria (logger.js)
Exibe no terminal todas as requisições recebidas em tempo real (útil para mostrar o funcionamento do servidor durante a apresentação):// src/middlewares/logger.js
function logger(req, res, next) {
console.log(`[${new Date().toISOString()}] ${req.method} ${req.url}`);
next();
}

module.exports = logger;

Lembrete: Registre app.use(logger); no src/server.js (logo abaixo de app.use(express.json());) e aplique validarId e validarPost (ou o validador do seu tema) nas rotas dentro de src/routes/.

## Rotas Principais da API
- `GET /...` - Lista todos os registros
- `GET /.../:id` - Busca um registro por ID
- `POST /...` - Cadastra um novo registro
- `PUT /.../:id` - Atualiza um registro
- `DELETE /.../:id` - Remove um registro
